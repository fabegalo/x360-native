# This project executes the same pinned CMake modules and complete compile/link
# checks as the caller. Even compiler discovery remains enabled in every worker.
cmake_minimum_required(VERSION 3.16)
include("${PROBE_CONTEXT}")
project(x360n_libc_probe LANGUAGES C)
include(CheckIncludeFile)
include(CheckSymbolExists)
file(WRITE "${CMAKE_BINARY_DIR}/results.cmake" "")
foreach(_name IN LISTS PROBE_NAMES)
    string(TOUPPER "${_name}" _variable)
    string(REGEX REPLACE "[./]" "_" _variable "${_variable}")
    set(_variable "LIBC_HAS_${_variable}")
    if(PROBE_KIND STREQUAL "headers")
        check_include_file("${_name}" "${_variable}")
    elseif(PROBE_KIND STREQUAL "symbols")
        check_symbol_exists("${_name}" "${PROBE_HEADERS}" "${_variable}")
    else()
        message(FATAL_ERROR "Unknown libc probe kind")
    endif()
    file(APPEND "${CMAKE_BINARY_DIR}/results.cmake" "set(${_variable} [==[${${_variable}}]==] CACHE INTERNAL \"Original CMake libc probe\" FORCE)\n")
endforeach()
