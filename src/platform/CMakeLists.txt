# Process supervision is shared by the CLI, desktop and preparation tools.
add_library(x360n_process STATIC ../supervisor/child_process.cpp)
if(UNIX)
  include(CheckSymbolExists)
  set(CMAKE_REQUIRED_DEFINITIONS -D_GNU_SOURCE)
  check_symbol_exists(posix_spawn_file_actions_addchdir_np "spawn.h" X360N_HAVE_SPAWN_CHDIR)
  if(NOT X360N_HAVE_SPAWN_CHDIR)
    message(FATAL_ERROR "Native process supervision needs posix_spawn_file_actions_addchdir_np (glibc >= 2.29 / macOS >= 10.15)")
  endif()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    check_symbol_exists(posix_spawn_file_actions_addclosefrom_np "spawn.h" X360N_HAVE_SPAWN_CLOSEFROM)
    if(NOT X360N_HAVE_SPAWN_CLOSEFROM)
      message(FATAL_ERROR "Native Linux process supervision requires glibc >= 2.34 for safe descriptor inheritance")
    endif()
  endif()
  unset(CMAKE_REQUIRED_DEFINITIONS)
  target_sources(x360n_process PRIVATE posix/process.cpp)
endif()
target_include_directories(x360n_process PUBLIC ../supervisor)
target_link_libraries(x360n_process PUBLIC fmt::fmt)
if(WIN32)
  target_link_libraries(x360n_process PUBLIC dbghelp)
endif()
find_package(Threads REQUIRED)
target_link_libraries(x360n_process PUBLIC Threads::Threads)
