cmake_minimum_required(VERSION 3.25)
project(build_isolation LANGUAGES CXX)
include("${X360N_ROOT}/cmake/sdk/x360n_sdk_backports.cmake")
include("${X360N_ROOT}/cmake/build/x360n_build_isolation.cmake")
x360n_guard_ninja()
if(TEST_GATE)
    add_custom_target(gate
        COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/gate.ps1" "${TEST_GATE}" "${CMAKE_BINARY_DIR}/gate-started"
        VERBATIM)
endif()
add_library(rexruntime SHARED runtime.cpp)
set(RUNTIME_REVISION 0 CACHE STRING "Runtime implementation revision (unchanged ABI)")
target_compile_definitions(rexruntime PRIVATE IDENTITY=${IDENTITY} RUNTIME_REVISION=${RUNTIME_REVISION})
set_target_properties(rexruntime PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/sdk/$<CONFIG>")
foreach(n RANGE 1 16)
    add_executable(consumer${n} main.cpp)
    target_compile_definitions(consumer${n} PRIVATE IDENTITY=${IDENTITY})
    target_link_libraries(consumer${n} PRIVATE rexruntime)
    x360n_stage_runtime_dlls(consumer${n})
endforeach()
