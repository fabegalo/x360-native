cmake_minimum_required(VERSION 3.25)
project(x360n_native_ppc_validation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT X360N_SHARED_SDK_EXPORT OR NOT X360_REXGLUE_DIR OR NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Use the already built native shared SDK, its pinned sources and Release")
endif()
get_filename_component(X360N_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(REXSDK_DIR "${X360_REXGLUE_DIR}")
include("${X360N_SHARED_SDK_EXPORT}")
enable_testing()
add_subdirectory("${X360_REXGLUE_DIR}/thirdparty/catch2" catch2)
list(APPEND CMAKE_MODULE_PATH "${X360_REXGLUE_DIR}/thirdparty/catch2/extras")
include(Catch)
include("${X360N_ROOT}/cmake/runtime/x360n_ppc.cmake")
include("${X360N_ROOT}/cmake/sdk/x360n_sdk_backports.cmake")
set(X360N_PPC_VALIDATE_GPU_STAGING ON)
add_subdirectory("${X360N_ROOT}/tests/ppc_smoke" smoke)
