# Shared automatic AOT host. No title-specific compatibility claims or overrides.
cmake_minimum_required(VERSION 3.25)
project(@TITLE@ LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(X360N_GENERATED_ROOT "@GENERATED@")
set(X360N_AOT_HOST_DISPATCH @HOST_DISPATCH@)
include(generated/x360n_rexglue.cmake)
include("@ROOT@/cmake/sdk/x360n_sdk_backports.cmake")
x360n_apply_sdk_backports("${REXSDK_DIR}")
add_executable(@TITLE@ WIN32 main.cpp "@ROOT@/titles/common/launch_handoff.cpp"
    "@ROOT@/titles/common/guest_fault_context.cpp")
target_include_directories(@TITLE@ PRIVATE "@ROOT@/titles/common")
rexglue_setup_target(@TITLE@ GPU_PLUGINS xenos)
if(X360N_AOT_HOST_DISPATCH)
    target_compile_definitions(@TITLE@ PRIVATE REX_HOST_FUNCTION_TABLE=1)
    if(TARGET @TITLE@_recomp)
        target_compile_definitions(@TITLE@_recomp PRIVATE REX_HOST_FUNCTION_TABLE=1)
    endif()
endif()
x360n_stage_runtime_dlls(@TITLE@ PLUGIN_TARGETS rexgpu-xenos)
