cmake_minimum_required(VERSION 3.25)
project(dash_1888 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Retail 1888 requires Release preparation")
endif()
include(generated/rexglue.cmake)
if(REXSDK_DIR)
    include(../../cmake/sdk/x360n_sdk_backports.cmake)
    x360n_apply_sdk_backports("${REXSDK_DIR}")
endif()
set(sources src/main.cpp src/boot_intro.cpp ../common/dashboard_assets.cpp ../common/launch_handoff.cpp
    ../../src/supervisor/sha256.cpp ../../src/supervisor/digest_batch.cpp)
if(WIN32)
    add_executable(dash_1888 WIN32 ${sources})
    target_link_libraries(dash_1888 PRIVATE bcrypt)
else()
    add_executable(dash_1888 ${sources})
endif()
rexglue_setup_target(dash_1888 GPU_PLUGINS xenos)
if(TARGET dash_1888_bootanim)
    target_sources(dash_1888_bootanim PRIVATE src/boot_intro_control.cpp)
endif()
include(../../cmake/runtime/x360n_dashboard_assets.cmake)
x360n_guard_dashboard_profile(dash_1888 "${CMAKE_CURRENT_SOURCE_DIR}/dash_1888_manifest.toml")
if(REXSDK_DIR)
    x360n_stage_runtime_dlls(dash_1888 PLUGIN_TARGETS rexgpu-xenos)
endif()
