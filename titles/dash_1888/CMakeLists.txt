cmake_minimum_required(VERSION 3.25)
project(dash_1888 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE STREQUAL "Release")
    message(FATAL_ERROR "Retail 1888 requires Release preparation")
endif()
include(generated/rexglue.cmake)
if(REXSDK_DIR)
    include(../../cmake/sdk/x360n_sdk_backports.cmake)
    x360n_apply_sdk_backports("${REXSDK_DIR}")
endif()
set(sources src/main.cpp src/boot_intro.cpp src/system_ui.cpp src/legacy_ui_requests.cpp
    ../common/dashboard_assets.cpp ../common/launch_handoff.cpp
    ../../src/supervisor/sha256.cpp ../../src/supervisor/digest_batch.cpp)
if(WIN32)
    add_executable(dash_1888 WIN32 ${sources})
    target_link_libraries(dash_1888 PRIVATE bcrypt)
else()
    add_executable(dash_1888 ${sources})
endif()
rexglue_setup_target(dash_1888 GPU_PLUGINS xenos)
if(TARGET dash_1888_bootanim)
    target_sources(dash_1888_bootanim PRIVATE src/boot_intro_control.cpp)
endif()

option(X360N_CODEGEN_BOOTSTRAP "Configure the producer before guest generation" OFF)
set(X360N_CODEGEN_MANIFESTS "")
foreach(module signin createprofile vk hud gamerprofile)
    set(manifest "${CMAKE_CURRENT_SOURCE_DIR}/${module}/${module}_1888_manifest.toml")
    string(APPEND X360N_CODEGEN_MANIFESTS "${manifest}\n")
    set(generated_dir "${CMAKE_CURRENT_SOURCE_DIR}/generated/${module}")
    if(NOT EXISTS "${generated_dir}/sources.cmake")
        if(NOT X360N_CODEGEN_BOOTSTRAP)
            message(FATAL_ERROR "Missing original 1888 ${module} AOT; run scripts/build-title.ps1 -Title dash_1888")
        endif()
        continue()
    endif()
    include("${generated_dir}/sources.cmake")
    add_custom_command(OUTPUT "${generated_dir}/codegen.build.stamp" ${GENERATED_SOURCES}
        COMMAND $<TARGET_FILE:rex::rexglue> codegen "${manifest}"
        DEPFILE "${generated_dir}/codegen.d"
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" VERBATIM)
    set(module_target "dash_1888_${module}")
    add_library(${module_target} SHARED ${GENERATED_SOURCES}
        src/system_ui_imports.cpp src/legacy_ui_requests.cpp src/system_ui_app_load.cpp)
    if(module STREQUAL "vk")
        target_sources(${module_target} PRIVATE src/vk_registration.cpp)
    endif()
    target_include_directories(${module_target} PRIVATE "${generated_dir}")
    target_link_libraries(${module_target} PRIVATE rex::runtime x360n_ppc_options)
    rexglue_apply_recomp_settings(${module_target} "${generated_dir}/dash_1888_pch.h")
    rexglue_configure_module_target(${module_target} HOST dash_1888)
    add_dependencies(dash_1888 ${module_target})
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/x360n-codegen-manifests.txt" "${X360N_CODEGEN_MANIFESTS}")
if(TARGET dash_1888_xam)
    target_sources(dash_1888_xam PRIVATE src/system_ui_memory.cpp)
    target_compile_definitions(dash_1888_xam PRIVATE REX_HOST_FUNCTION_TABLE=1)
endif()
include(../../cmake/runtime/x360n_dashboard_assets.cmake)
x360n_guard_dashboard_profile(dash_1888 "${CMAKE_CURRENT_SOURCE_DIR}/dash_1888_manifest.toml")
if(REXSDK_DIR)
    x360n_stage_runtime_dlls(dash_1888 PLUGIN_TARGETS rexgpu-xenos)
endif()
